   485→  /* ============================
   486→     IMAGE UPLOAD
   487→     ============================ */

   489→  async function handleImageUpload(
   490→    e: React.ChangeEvent<HTMLInputElement>,
   491→    type: "department" | "gallery",
   492→    index: number
   493→  ) {
   494→    const file = e.target.files?.[0];
   495→    if (!file) return;
   496→
   497→    const formData = new FormData();
   498→    formData.append("file", file);
   499→    formData.append("folder", type === "department" ? "departments" : type);
   500→
   501→    try {
   502→      const res = await fetch("/api/admin/upload", {
   503→        method: "POST",
   504→        body: formData,
   505→      });
   506→      if (res.ok) {
   507→        const data = await res.json();
   508→        const url = data.url;
   509→
   510→        /* Update local state first for instant preview */
   511→        if (type === "department") {
   512→          const item = departments[index];
   513→          setDepartments((prev) =>
   514→            prev.map((d, i) => (i === index ? { ...d, image: url } : d))
   515→          );
   516→          /* Auto-save to database so image persists */
   517→          if (item && !item.id.startsWith("new-")) {
   518→            await fetch("/api/admin/departments", {
   519→              method: "PUT",
   520→              headers: { "Content-Type": "application/json" },
   521→              body: JSON.stringify({ ...item, image: url }),
   522→            });
   523→          }
   524→        } else if (type === "gallery") {
   525→          const item = galleryItems[index];
   526→          setGalleryItems((prev) =>
   527→            prev.map((g, i) => (i === index ? { ...g, image: url } : g))
   528→          );
   529→          /* Auto-save to database */
   530→          if (item && !item.id.startsWith("new-")) {
   531→            await fetch("/api/admin/gallery", {
   532→              method: "PUT",
   533→              headers: { "Content-Type": "application/json" },
   534→              body: JSON.stringify({ ...item, image: url }),
   535→            });
   536→          }
   537→        }
   538→
   539→        /* Refresh public-facing site content */
   540→        await refetch();
   541→        toast.success("وێنەکە بارکرا و تۆمارکرا");
   542→      } else {
   543→        toast.error("هەڵەیەک ڕوویدا لە بارکردنی وێنە");
   544→      }
   545→    } catch {
   546→      toast.error("هەڵەیەک ڕوویدا لە بارکردنی وێنە");
   547→    }
   548→
   549→    /* Reset file input */
   550→    e.target.value = "";
   551→  }

   553→  function triggerUpload(type: "department" | "gallery", index: number) {
   554→    /* Use a short timeout to ensure the file input is mounted before clicking */
   555→    setTimeout(() => {
   556→      const refMap =
   557→        type === "department"
   558→          ? deptFileRefs
   559→          : galFileRefs;
   560→      refMap.current[index]?.click();
   561→    }, 50);
   562→  }

   564→  /* Hidden file input - each item gets its own unique input element
   565→     so that clicking upload on item #1 always uploads to item #1 */
   566→  function getRefCallback(
   567→    type: "department" | "gallery",
   568→    index: number
   569→  ) {
   570→    const refMap =
   571→      type === "department"
   572→        ? deptFileRefs
   573→        : galFileRefs;
   574→    return (el: HTMLInputElement | null) => {
   575→      refMap.current[index] = el;
   576→    };
   577→  }
   578→
   579→  const hiddenFileInput = (
   580→    type: "department" | "gallery",
   581→    index: number
   582→  ) => {
   583→    return (
   584→      <input
   585→        key={`${type}-${index}`}
   586→        ref={getRefCallback(type, index)}
   587→        type="file"
   588→        accept="image/*"
   589→        className="hidden"
   590→        onChange={(e) => handleImageUpload(e, type, index)}
   591→      />
   592→    );
   593→  };

  1240→  /* Handle multi-image upload for an archive item */
  1241→  async function handleArchiveImageUpload(
  1242→    e: React.ChangeEvent<HTMLInputElement>,
  1243→    archiveIdx: number
  1244→  ) {
  1245→    const files = e.target.files;
  1246→    if (!files || files.length === 0) return;
  1247→
  1248→    const item = archiveItems[archiveIdx];
  1249→    if (!item) return;
  1250→
  1251→    try {
  1252→      for (let i = 0; i < files.length; i++) {
  1253→        const file = files[i];
  1254→        const formData = new FormData();
  1255→        formData.append("file", file);
  1256→        formData.append("folder", "archive");
  1257→
  1258→        const res = await fetch("/api/admin/upload", {
  1259→          method: "POST",
  1260→          body: formData,
  1261→        });
  1262→
  1263→        if (res.ok) {
  1264→          const data = await res.json();
  1265→          const url = data.url;
  1266→
  1267→          /* Save image to database via archive-images API */
  1268→          if (!item.id.startsWith("new-")) {
  1269→            await fetch("/api/admin/archive-images", {
  1270→              method: "POST",
  1271→              headers: { "Content-Type": "application/json" },
  1272→              body: JSON.stringify({
  1273→                archiveItemId: item.id,
  1274→                url,
  1275→                order: (item.images?.length || 0) + i,
  1276→              }),
  1277→            });
  1278→          }
  1279→
  1280→          /* Update local state */
  1281→          const newImage: ArchiveImage = {
  1282→            id: `temp-${Date.now()}-${i}`,
  1283→            url,
  1284→            order: (item.images?.length || 0) + i,
  1285→            archiveItemId: item.id,
  1286→          };
  1287→          setArchiveItems((prev) =>
  1288→            prev.map((a, idx) =>
  1289→              idx === archiveIdx
  1290→                ? { ...a, images: [...(a.images || []), newImage] }
  1291→                : a
  1292→            )
  1293→          );
  1294→        }
  1295→      }
  1296→
  1297→      await refetch();
  1298→      toast.success(
  1299→        files.length === 1
  1300→          ? "وێنەکە بارکرا و تۆمارکرا"
  1301→          : `${files.length} وێنە بارکران و تۆمارکران`
  1302→      );
  1303→    } catch {
  1304→      toast.error("هەڵەیەک ڕوویدا لە بارکردنی وێنە");
  1305→    }
  1306→
  1307→    e.target.value = "";
  1308→  }

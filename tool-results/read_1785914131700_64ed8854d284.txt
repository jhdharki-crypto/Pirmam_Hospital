   479→  /* ============================
   480→     IMAGE UPLOAD
   481→     ============================ */
   482→
   483→  async function handleImageUpload(
   484→    e: React.ChangeEvent<HTMLInputElement>,
   485→    type: "department" | "gallery" | "archive",
   486→    index: number
   487→  ) {
   488→    const file = e.target.files?.[0];
   489→    if (!file) return;
   490→
   491→    const formData = new FormData();
   492→    formData.append("file", file);
   493→    formData.append("folder", type === "department" ? "departments" : type);
   494→
   495→    try {
   496→      const res = await fetch("/api/admin/upload", {
   497→        method: "POST",
   498→        body: formData,
   499→      });
   500→      if (res.ok) {
   501→        const data = await res.json();
   502→        const url = data.url;
   503→        if (type === "department") {
   504→          setDepartments((prev) =>
   505→            prev.map((d, i) => (i === index ? { ...d, image: url } : d))
   506→          );
   507→        } else if (type === "gallery") {
   508→          setGalleryItems((prev) =>
   509→            prev.map((g, i) => (i === index ? { ...g, image: url } : g))
   510→          );
   511→        } else {
   512→          setArchiveItems((prev) =>
   513→            prev.map((a, i) => (i === index ? { ...a, image: url } : a))
   514→          );
   515→        }
   516→        toast.success("وێنەکە بارکرا");
   517→      } else {
   518→        toast.error("هەڵەیەک ڕوویدا لە بارکردنی وێنە");
   519→      }
   520→    } catch {
   521→      toast.error("هەڵەیەک ڕوویدا لە بارکردنی وێنە");
   522→    }
   523→
   524→    /* Reset file input */
   525→    e.target.value = "";
   526→  }
